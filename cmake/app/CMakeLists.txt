cmake_minimum_required(VERSION 3.22)
project(KomodoOcean LANGUAGES C CXX)
get_filename_component(KMD_CMAKE_DIR "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
get_filename_component(KMD_REPOSITORY "${KMD_CMAKE_DIR}/.." REALPATH)
include("${KMD_CMAKE_DIR}/Options.cmake")
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR CMAKE_CROSSCOMPILING OR
    NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "Currently supported: native Linux x86_64")
endif()
file(REAL_PATH "${CMAKE_BINARY_DIR}" binary_path)
if(binary_path STREQUAL KMD_REPOSITORY OR binary_path MATCHES "^${KMD_REPOSITORY}/(src|depends|cmake)(/|$)")
  message(FATAL_ERROR "Use a separate application build directory")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Consensus code uses assertions in release builds as well.
set(CMAKE_C_FLAGS_RELEASE "-O2 -g0")
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -g0")
set(CMAKE_C_FLAGS_RELWITHDEBINFO "-O2 -g")
set(CMAKE_CXX_FLAGS_RELWITHDEBINFO "-O2 -g")
set(CMAKE_C_FLAGS_MINSIZEREL "-Os")
set(CMAKE_CXX_FLAGS_MINSIZEREL "-Os")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(KMD_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" CACHE PATH "Program output directory")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${KMD_OUTPUT_DIRECTORY}")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
include("${KMD_CMAKE_DIR}/Snapshot.cmake")
kmd_snapshot("${KMD_CMAKE_DIR}/src-inputs.txt" "${CMAKE_BINARY_DIR}/source")
set(SRC "${CMAKE_BINARY_DIR}/source/src")
include("${KMD_CMAKE_DIR}/dependencies/ImportDepends.cmake")
include("${KMD_CMAKE_DIR}/checks/Configure.cmake")
include("${KMD_CMAKE_DIR}/targets/Sources.cmake")
include("${KMD_CMAKE_DIR}/targets/Libraries.cmake")
include("${KMD_CMAKE_DIR}/targets/Programs.cmake")
if(KMD_BUILD_GUI)
  include("${KMD_CMAKE_DIR}/targets/Qt.cmake")
endif()
